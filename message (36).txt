-- ========== LECTOR ==========
local function extractBlock(text, key)
	local s, e = text:find("%f[%w_]" .. key .. "%s*=%s*{")
	if not s then return nil end
	local depth, i = 1, e + 1
	while i <= #text and depth > 0 do
		local c = text:sub(i, i)
		if c == "{" then depth += 1 elseif c == "}" then depth -= 1 end
		i += 1
	end
	return text:sub(e + 1, i - 2)
end

local function toCFrame(nums)
	local t = {}
	for part in nums:gmatch("[^,]+") do
		local n = tonumber(part)
		if n then t[#t + 1] = n end
	end
	if #t == 12 then return CFrame.new(table.unpack(t)) end
	if #t == 3 then return CFrame.new(t[1], t[2], t[3]) end
end

local function parsePoses(block)
	local poses = {}
	if not block then return poses end
	for name, nums in block:gmatch('%["([^"]+)"%]%s*=%s*CFrame%.new%(([^%)]*)%)') do
		poses[name] = toCFrame(nums)
	end
	for name, nums in block:gmatch('([%a_][%w_]*)%s*=%s*CFrame%.new%(([^%)]*)%)') do
		poses[name] = toCFrame(nums)
	end
	return poses
end

local function parseOffsets(block)
	local o = {}
	if not block then return o end
	for name, v in block:gmatch('%["([^"]+)"%]%s*=%s*([%-%d%.]+)') do o[name] = tonumber(v) end
	for name, v in block:gmatch('([%a_][%w_]*)%s*=%s*([%-%d%.]+)') do o[name] = tonumber(v) end
	return o
end

local function parseAnimation(text)
	local anim = {
		offsets = parseOffsets(extractBlock(text, "offsets")),
		fullOffset = tonumber(text:match("fullOffset%s*=%s*([%-%d%.]+)")) or 0,
		speed = tonumber(text:match("%f[%w_]speed%s*=%s*([%-%d%.]+)")) or 1,
		a = parsePoses(extractBlock(text, "a")),
		b = parsePoses(extractBlock(text, "b")),
	}
	if next(anim.a) == nil then return nil end
	return anim
end

local ORDER = {"RootJoint", "Neck", "Left Shoulder", "Right Shoulder", "Left Hip", "Right Hip"}
local conn, stepConn, oldProps
local removed = {}

local function stop()
	if conn then conn:Disconnect() conn = nil end
	if stepConn then stepConn:Disconnect() stepConn = nil end
	local char = player.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if hrp and oldProps ~= nil then hrp.CustomPhysicalProperties = oldProps end
	oldProps = nil
	for _, r in ipairs(removed) do
		if r.parent and r.parent.Parent then r.clone.Parent = r.parent end
	end
	removed = {}
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if hum then
		pcall(function()
			hum.RequiresNeck = true
			hum:SetStateEnabled(Enum.HumanoidStateType.Dead, true)
		end)
	end
end

local function play(anim)
	stop()
	local char = player.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not (hum and hrp and char:FindFirstChild("Torso")) then
		return false, "Necesitas un personaje R6"
	end

	local usaCuello = USAR_CUELLO and anim.a["Neck"] ~= nil
	if usaCuello then
		pcall(function()
			hum.RequiresNeck = false
			hum:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
		end)
	end

	local joints = {}
	for _, name in ipairs(ORDER) do
		if anim.a[name] and (name ~= "Neck" or usaCuello) then
			for _, d in ipairs(char:GetDescendants()) do
				if d:IsA("Motor6D") and d.Name == name then
					if d.Part0 and d.Part1 then
						joints[#joints + 1] = {
							name = name, part0 = d.Part0, part1 = d.Part1,
							c0 = d.C0, c1inv = d.C1:Inverse(),
						}
						removed[#removed + 1] = {clone = d:Clone(), parent = d.Parent}
						d:Destroy()
					end
					break
				end
			end
		end
	end

	-- el arreglo de peso y colisiones solo hace falta si se mueve el RootJoint
	if anim.a["RootJoint"] then
		oldProps = hrp.CustomPhysicalProperties
		hrp.CustomPhysicalProperties = PhysicalProperties.new(DENSIDAD_HRP, 0.3, 0, 1, 1)
	end

	local t0 = os.clock()
	conn = RunService.Heartbeat:Connect(function()
		local el = os.clock() - t0
		for _, jd in ipairs(joints) do
			local a = anim.a[jd.name]
			local b = anim.b[jd.name] or a
			local off = anim.offsets[jd.name] or 0
			local phase = el * anim.speed + anim.fullOffset + off
			local alpha = 0.5 - 0.5 * math.cos(phase * math.pi * 2)
			jd.part1.CFrame = jd.part0.CFrame * jd.c0 * a:Lerp(b, alpha) * jd.c1inv
			jd.part1.AssemblyLinearVelocity = Vector3.zero
			jd.part1.AssemblyAngularVelocity = Vector3.zero
		end
	end)
	return true
end

player.CharacterAdded:Connect(function()
	if conn then conn:Disconnect() conn = nil end
	if stepConn then stepConn:Disconnect() stepConn = nil end
	oldProps = nil
	removed = {}
end)

	for _, p in ipairs(char:GetChildren()) do
		if p:IsA("BasePart") then p.CanCollide = false end
	end
end)

local function normalizeId(id)
	if type(id) == "number" then
		if id == 0 then return nil end
		return "rbxassetid://" .. id
	end
	if type(id) == "string" and id ~= "" and id ~= "0" then
		if id:match("^%d+$") then

		end
		return id
	end
	return nil
end

	local ok, res = pcall(function()
		return InsertService:CreateMeshPartAsync(meshId, Enum.CollisionFidelity.Box, Enum.RenderFidelity.Automatic)
	end)
	if ok and res then
		mp = res
	else
		-- 2) directo (ejecutores): el mesh carga después, se corrige el tamaño en buildVest
		pcall(function()
			local m = Instance.new("MeshPart")
			m.MeshId = meshId
			mp = m
		end)
	end
	if not mp then
		warn("[Morph] No se pudo crear el MeshPart (meshId " .. tostring(meshId) .. ")")
		return nil
	end
	local tex = normalizeId(cfg.textureId)
	if tex then
		pcall(function() mp.TextureID = tex end)
	end
	return mp
end

-- Físicas: suelda el mesh al torso con un Weld cuyo C0 se mueve cada frame con un muelle.
local function attachJiggle(mp, torso, cfg)
	local offset = cfg.offset or Vector3.new(0, 0, 0)
	local rot = cfg.rotation or Vector3.new(0, 0, 0)
	local size = cfg.size or Vector3.new(1, 1, 1)
	local baseCF = CFrame.new(offset) * CFrame.Angles(math.rad(rot.X), math.rad(rot.Y), math.rad(rot.Z))
	-- el mesh se balancea desde un punto de su parte de arriba (como si colgara del pecho)
	local pivot = offset + Vector3.new(0, cfg.pivotY or size.Y * 0.4, 0)

	local weld = Instance.new("Weld")
	weld.Part0 = torso
	weld.Part1 = mp
	weld.C0 = baseCF
	weld.C1 = CFrame.new()
	weld.Parent = mp

	local P = PHYS
	local pitch, pitchV = 0, 0
	local roll, rollV = 0, 0
	local move, moveV = 0, 0
	local lastPos = torso.Position
	local lastVel = Vector3.new(0, 0, 0)

	local conn
	conn = RunService.RenderStepped:Connect(function(dt)
		if not mp.Parent or not torso.Parent then
			conn:Disconnect()
			return
		end
		dt = math.clamp(dt, 1 / 240, 1 / 30)

		local pos = torso.Position
		local vel = (pos - lastPos) / dt
		local dv = vel - lastVel
		lastPos, lastVel = pos, vel

		local cf = torso.CFrame
		local dUp = math.clamp(dv:Dot(cf.UpVector), -P.maxImpulse, P.maxImpulse)
		local dRight = math.clamp(dv:Dot(cf.RightVector), -P.maxImpulse, P.maxImpulse)
		pitchV += dUp * P.pitchGain
		rollV += dRight * P.rollGain
		moveV += dUp * P.moveGain

		pitchV += (-P.stiffness * pitch - P.damping * pitchV) * dt
		rollV += (-P.stiffness * roll - P.damping * rollV) * dt
		moveV += (-P.stiffness * move - P.damping * moveV) * dt
		pitch = math.clamp(pitch + pitchV * dt, -P.maxPitch, P.maxPitch)
		roll = math.clamp(roll + rollV * dt, -P.maxRoll, P.maxRoll)
		move = math.clamp(move + moveV * dt, -P.maxMove, P.maxMove)

		local jiggle = CFrame.new(pivot)
			* CFrame.Angles(math.rad(pitch), 0, math.rad(roll))
			* CFrame.new(0, move, 0)
			* CFrame.new(-pivot)
		weld.C0 = jiggle * baseCF
	end)
end

local SHIRT_TEMPLATE = { x = 231, y = 74, w = 128, h = 128 } -- dónde está el pecho (frente) dentro de la plantilla de camiseta de 585x559

-- Modo "project": calcomanía de un trozo de la camiseta (SurfaceGui + ImageLabel recortada)
local function attachShirtProject(mp, character, cfg)
	local crop = cfg.shirtCrop or { 0.6, 1 }
	local T = SHIRT_TEMPLATE
	local faces = cfg.decalFace and { cfg.decalFace } or {
		Enum.NormalId.Front, Enum.NormalId.Back, Enum.NormalId.Left,
		Enum.NormalId.Right, Enum.NormalId.Top, Enum.NormalId.Bottom,
	}
	local guis, imgs = {}, {}
	for _, face in ipairs(faces) do
		local sg = Instance.new("SurfaceGui")
		sg.Name = "ShirtDecal"
		sg.Face = face
		sg.LightInfluence = 0
		sg.ResetOnSpawn = false
		sg.Enabled = false
		sg.Adornee = mp
		sg.Parent = mp
		local img = Instance.new("ImageLabel")
		img.BackgroundTransparency = 1
		img.Size = UDim2.fromScale(1, 1)
		img.ScaleType = Enum.ScaleType.Stretch
		img.ImageRectOffset = Vector2.new(T.x, T.y + T.h * crop[1])
		img.ImageRectSize = Vector2.new(T.w, T.h * (crop[2] - crop[1]))
		img.Parent = sg
		table.insert(guis, sg)
		table.insert(imgs, img)
	end
	task.spawn(function()
		local last = nil
		while mp.Parent do
			local shirt = character:FindFirstChildOfClass("Shirt")
			local tex = shirt and shirt.ShirtTemplate or ""
			if tex ~= last then
				last = tex
				for i, img in ipairs(imgs) do
					img.Image = tex
					guis[i].Enabled = tex ~= ""
				end
			end
			task.wait(0.5)
		end
	end)
end

-- Camiseta en el mesh: el mesh base se queda del color de la piel y encima va una copia un pelín más grande con la
-- camiseta como SurfaceAppearance (AlphaMode = Transparency), así las zonas transparentes de la camiseta dejan ver la piel.
-- Se actualiza sola si el personaje cambia de camiseta.
local function attachShirtTexture(mp, character)
	local ov = mp:Clone()
	for _, c in ipairs(ov:GetChildren()) do c:Destroy() end
	ov.Name = "ShirtOverlay"
	ov.TextureID = ""
	ov.Anchored = false
	ov.CanCollide = false
	ov.CanTouch = false
	ov.CanQuery = false
	ov.Massless = true
	ov.Transparency = 1
	local sa = Instance.new("SurfaceAppearance")
	sa.AlphaMode = Enum.AlphaMode.Transparency
	sa.Parent = ov
	local weld = Instance.new("Weld")
	weld.Part0 = mp
	weld.Part1 = ov
	weld.Parent = ov
	ov.Parent = mp

	task.spawn(function()
		local last = nil
		while mp.Parent do
			local shirt = character:FindFirstChildOfClass("Shirt")
			local tex = shirt and shirt.ShirtTemplate or ""
			if tex ~= last then
				last = tex
				pcall(function() sa.ColorMap = tex end)
				ov.Transparency = (tex == "") and 1 or 0
			end
			-- la copia sigue el tamaño del mesh (que se corrige cuando carga) un 1.5% más grande
			local want = mp.Size * 1.015
			if (ov.Size - want).Magnitude > 0.001 then ov.Size = want end
			task.wait(0.1)
		end
	end)
end

local function buildVest(torso, folder, tipo)
	local list = MORPHS[tipo]
	if not list then return end
	local character = torso.Parent
	for i, cfg in ipairs(list) do
		local mp = makeMeshPart(cfg, i)
		if mp then
			local rot = cfg.rotation or Vector3.new(0, 0, 0)
			mp.Name = "Mesh" .. i
			local wantSize = cfg.size or Vector3.new(1, 1, 1)
			mp.Size = wantSize
			mp.Anchored = false
			mp.CanCollide = false
			mp.CanTouch = false
			mp.CanQuery = false
			mp.Massless = true
			mp.Material = Enum.Material.SmoothPlastic

			local col = resolverColor(cfg.color)
			if cfg.color and not col then warn("[Morph] Color desconocido: " .. tostring(cfg.color)) end
			mp.Color = col or torso.Color -- sin color propio: se adapta al color de piel del torso

			mp.CFrame = torso.CFrame
				* CFrame.new(cfg.offset or Vector3.new(0, 0, 0))
				* CFrame.Angles(math.rad(rot.X), math.rad(rot.Y), math.rad(rot.Z))
			mp.Parent = folder

			if cfg.shirtTexture == "project" then
				attachShirtProject(mp, character, cfg)
			elseif cfg.shirtTexture then
				attachShirtTexture(mp, character)
			end

			if cfg.physics then
				attachJiggle(mp, torso, cfg)
			else
				local weld = Instance.new("WeldConstraint")
				weld.Part0 = torso
				weld.Part1 = mp
				weld.Parent = mp
			end

			-- Si el mesh cargó DESPUÉS de ponerle el Size, Roblox calcula la escala sobre un tamaño inicial
			-- erróneo y queda gigante. Por eso, cuando cargue: primero se pone su tamaño original (escala 1)
			-- y después el tamaño que quieres.
			task.spawn(function()
				local function native()
					local ok, v = pcall(function() return mp.MeshSize end)
					return ok and v or Vector3.new(0, 0, 0)
				end
				local t = 0
				while mp.Parent and native().Magnitude == 0 and t < 6 do
					t += task.wait(0.1)
				end
				if mp.Parent then
					local ms = native()
					if ms.Magnitude > 0 then
						mp.Size = ms
					end
					mp.Size = wantSize
				end
			end)
		end
	end
end

local function quitarMorph(plr)
	local c = plr.Character
	local old = c and c:FindFirstChild(MORPH_NAME)
	if old then
		old:Destroy()
		return true
	end
	return false
end

local function applyMorph(plr, tipo)
	local char = plr.Character
	local torso = char and (char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso"))
	if not torso then return false, "Ese jugador no tiene torso" end

	local old = char:FindFirstChild(MORPH_NAME)
	if old then old:Destroy() end
	if tipo == "" then return true, "Morph quitado a " .. plr.Name end

	local folder = Instance.new("Folder")
	folder.Name = MORPH_NAME
	folder.Parent = char
	buildVest(torso, folder, tipo)
	return true, "Morph " .. tipo .. " puesto a " .. plr.Name
end

-- ========== BÚSQUEDA CON ABREVIACIONES ==========
local function subsequence(needle, hay)
	local i = 1
	for c in hay:gmatch(".") do
		if c == needle:sub(i, i) then
			i += 1
			if i > #needle then return true end
		end
	end
	return false
end

local function scoreName(query, name)
	name = name:lower()
	if name == query then return 100 end
	if name:sub(1, #query) == query then return 80 end -- empieza por...
	if name:find(query, 1, true) then return 60 end    -- contiene...
	if subsequence(query, name) then return 40 end     -- letras en orden (abreviación)
	return 0
end

local function findPlayer(text)
	local q = (text:lower():gsub("%s", ""))
	if q == "" then return nil end
	local best, bestScore = nil, 0
	for _, p in ipairs(Players:GetPlayers()) do
		local s = math.max(scoreName(q, p.Name), scoreName(q, p.DisplayName))
		if s > bestScore or (s == bestScore and s > 0 and best and #p.Name < #best.Name) then
			best, bestScore = p, s
		end
	end
	return best
end
